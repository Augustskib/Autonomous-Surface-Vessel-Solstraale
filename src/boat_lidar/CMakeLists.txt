cmake_minimum_required(VERSION 3.16)
project(boat_lidar)
find_package(ament_cmake REQUIRED)
option(ASV_BUILD_QUANERGY "Build the hardware adapter against the separately installed Quanergy SDK" OFF)
if(ASV_BUILD_QUANERGY)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(pcl_conversions REQUIRED)
  find_package(PCL REQUIRED COMPONENTS common)
  find_package(Boost REQUIRED COMPONENTS program_options system)
  find_package(QuanergyClient REQUIRED CONFIG)
  add_executable(quanergy_m8_node src/quanergy_m8_node.cpp)
  target_compile_features(quanergy_m8_node PRIVATE cxx_std_17)
  target_include_directories(quanergy_m8_node SYSTEM PRIVATE ${QUANERGY_CLIENT_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
  target_link_libraries(quanergy_m8_node ${QUANERGY_CLIENT_LIBRARIES} ${PCL_LIBRARIES})
  ament_target_dependencies(quanergy_m8_node rclcpp sensor_msgs std_msgs pcl_conversions)
  install(TARGETS quanergy_m8_node DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
